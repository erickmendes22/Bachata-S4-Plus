cmake_minimum_required(VERSION 3.22)
project(bachata_fsr31_core025 LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
set(FFX_VK_PORTABLE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FFX_VK_PORTABLE_BUILD_PROBE OFF CACHE BOOL "" FORCE)
set(FFX_VK_PORTABLE_BUILD_FSR3_HOST ON CACHE BOOL "" FORCE)
set(FFX_VK_PORTABLE_BUILD_FSR3_VK_BACKEND ON CACHE BOOL "" FORCE)
set(FFX_VK_PORTABLE_BUILD_FSR3_3_1_5_SCAFFOLD OFF CACHE BOOL "" FORCE)
set(FFX_VK_PORTABLE_BUILD_FSR4_V07_VULKAN OFF CACHE BOOL "" FORCE)
set(FFX_VK_PORTABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(BS4_FSR_SDK_DIR "" CACHE PATH "Pinned FSR-Vulkan c64f093 with Android portability patches")
set(BS4_GENERATED_DIR "" CACHE PATH "Compiled camera and flow shader headers")
add_subdirectory("${BS4_FSR_SDK_DIR}" sdk)
find_package(Vulkan REQUIRED)
add_library(bachata_fsr31 SHARED ../fsr31-026/bridge.cpp ../fsr31-026/vulkan_capture.cpp
 ../fsr31-026/port_camera_probe.cpp ../fsr31-026/port_ui_redirect.cpp ../fsr31-026/temporal_generic.cpp)
target_include_directories(bachata_fsr31 PRIVATE "${BS4_GENERATED_DIR}")
target_compile_definitions(bachata_fsr31 PRIVATE _GNU_SOURCE=1)
target_link_libraries(bachata_fsr31 PRIVATE ffx-vulkan::portable Vulkan::Vulkan dl)
